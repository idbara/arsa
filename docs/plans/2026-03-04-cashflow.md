### Task 2: Create cashflow server loader

**Files:**
- Create: `arsa-app/src/routes/cashflow/+page.server.ts`

**Context:** This loader fetches two datasets server-side: (1) current month stats grouped by type and category, (2) 6-month trend grouped by month and type. It also loads category names for labeling.

**Step 1: Create the server loader**

```typescript
import type { PageServerLoad } from './$types';

export const load: PageServerLoad = async ({ locals: { supabase, safeGetSession }, url }) => {
	const { user } = await safeGetSession();
	if (!user) return { stats: { income: 0, expense: 0 }, trend: [], categoryBreakdown: [], categories: [] };

	const { data: profile } = await supabase
		.from('users')
		.select('family_id')
		.eq('id', user.id)
		.single();

	if (!profile?.family_id)
		return { stats: { income: 0, expense: 0 }, trend: [], categoryBreakdown: [], categories: [] };

	const familyId = profile.family_id;

	// Parse month from URL params, default to current month
	const monthParam = url.searchParams.get('month');
	let year: number, month: number;
	if (monthParam && /^\d{4}-\d{2}$/.test(monthParam)) {
		[year, month] = monthParam.split('-').map(Number);
	} else {
		const now = new Date();
		year = now.getFullYear();
		month = now.getMonth() + 1;
	}

	const monthStart = `${year}-${String(month).padStart(2, '0')}-01`;
	const lastDay = new Date(year, month, 0).getDate();
	const monthEnd = `${year}-${String(month).padStart(2, '0')}-${String(lastDay).padStart(2, '0')}`;

	// Query 1: Current month stats by type and category
	const { data: monthTx } = await supabase
		.from('transactions')
		.select('type, category_id, amount')
		.eq('family_id', familyId)
		.gte('date', monthStart)
		.lte('date', monthEnd);

	let income = 0;
	let expense = 0;
	const categoryMap = new Map<string, { total: number; count: number }>();

	for (const tx of monthTx ?? []) {
		const amt = Number(tx.amount);
		if (tx.type === 'in') income += amt;
		if (tx.type === 'out') {
			expense += amt;
			if (tx.category_id) {
				const existing = categoryMap.get(tx.category_id) ?? { total: 0, count: 0 };
				existing.total += amt;
				existing.count += 1;
				categoryMap.set(tx.category_id, existing);
			}
		}
	}

	// Load category names
	const { data: categories } = await supabase
		.from('categories')
		.select('id, name')
		.eq('family_id', familyId);

	const catNameMap = new Map((categories ?? []).map((c) => [c.id, c.name]));

	const categoryBreakdown = Array.from(categoryMap.entries())
		.map(([id, data]) => ({
			id,
			name: catNameMap.get(id) ?? 'Uncategorized',
			total: data.total,
			count: data.count,
			percentage: expense > 0 ? Math.round((data.total / expense) * 100) : 0
		}))
		.sort((a, b) => b.total - a.total);

	// Cap at 8 categories, merge rest into "Lainnya"
	let finalBreakdown = categoryBreakdown;
	if (categoryBreakdown.length > 8) {
		const top8 = categoryBreakdown.slice(0, 8);
		const rest = categoryBreakdown.slice(8);
		const restTotal = rest.reduce((s, c) => s + c.total, 0);
		const restCount = rest.reduce((s, c) => s + c.count, 0);
		top8.push({
			id: '__other__',
			name: 'Lainnya',
			total: restTotal,
			count: restCount,
			percentage: expense > 0 ? Math.round((restTotal / expense) * 100) : 0
		});
		finalBreakdown = top8;
	}

	// Query 2: 6-month trend
	const trendStart = new Date(year, month - 7, 1);
	const trendStartStr = trendStart.toISOString().split('T')[0];

	const { data: trendTx } = await supabase
		.from('transactions')
		.select('type, amount, date')
		.eq('family_id', familyId)
		.gte('date', trendStartStr)
		.lte('date', monthEnd);

	const trendMap = new Map<string, { income: number; expense: number }>();

	for (const tx of trendTx ?? []) {
		const txDate = new Date(tx.date + 'T00:00:00');
		const key = `${txDate.getFullYear()}-${String(txDate.getMonth() + 1).padStart(2, '0')}`;
		const existing = trendMap.get(key) ?? { income: 0, expense: 0 };
		const amt = Number(tx.amount);
		if (tx.type === 'in') existing.income += amt;
		if (tx.type === 'out') existing.expense += amt;
		trendMap.set(key, existing);
	}

	// Build 6-month array (always 6 entries, fill 0 for missing months)
	const trend = [];
	for (let i = 5; i >= 0; i--) {
		const d = new Date(year, month - 1 - i, 1);
		const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
		const label = d.toLocaleDateString('id-ID', { month: 'short' });
		const data = trendMap.get(key) ?? { income: 0, expense: 0 };
		trend.push({ month: key, label, ...data });
	}

	return {
		stats: { income, expense },
		trend,
		categoryBreakdown: finalBreakdown,
		currentMonth: `${year}-${String(month).padStart(2, '0')}`,
		monthLabel: new Date(year, month - 1).toLocaleDateString('id-ID', { month: 'long', year: 'numeric' })
	};
};
```

<script lang="ts">
	import { goto } from '$app/navigation';
	import { resolve } from '$app/paths';

	let { data } = $props();

	function navigateMonth(delta: number) {
		const [y, m] = data.currentMonth.split('-').map(Number);
		const d = new Date(y, m - 1 + delta, 1);

	id: string;
	date: string;
	merchant: string | null;
	amount: number;
	is_private: boolean;
}>>([]);

async function toggleCategory(categoryId: string) {
	if (expandedCategory === categoryId) {
		expandedCategory = null;
		return;

	expandedCategory = categoryId;
	loadingCategory = true;

	const res = await fetch(
		resolve(`/api/cashflow/transactions?category_id=${categoryId}&month=${data.currentMonth}`)

	const json = await res.json();
	categoryTransactions = json.transactions ?? [];
	loadingCategory = false;
}

### Task 7: Create API route for category transactions

**Files:**
- Create: `arsa-app/src/routes/api/cashflow/transactions/+server.ts`

**Context:** This API route serves transaction details for a specific category and month. Called on-demand when user expands a category row. Respects private transaction rules.

**Step 1: Create the API route**

```typescript
import { json } from '@sveltejs/kit';
import type { RequestHandler } from './$types';

export const GET: RequestHandler = async ({ url, locals: { supabase, safeGetSession } }) => {
	const { user } = await safeGetSession();
	if (!user) return json({ transactions: [] }, { status: 401 });

	const { data: profile } = await supabase
		.from('users')
		.select('family_id')
		.eq('id', user.id)
		.single();

	if (!profile?.family_id) return json({ transactions: [] }, { status: 400 });

	const categoryId = url.searchParams.get('category_id');
	const monthParam = url.searchParams.get('month');

	if (!categoryId || !monthParam || !/^\d{4}-\d{2}$/.test(monthParam)) {
		return json({ transactions: [] }, { status: 400 });
	}

	const [year, month] = monthParam.split('-').map(Number);
	const monthStart = `${year}-${String(month).padStart(2, '0')}-01`;
	const lastDay = new Date(year, month, 0).getDate();
	const monthEnd = `${year}-${String(month).padStart(2, '0')}-${String(lastDay).padStart(2, '0')}`;

	const { data: transactions } = await supabase
		.from('transactions')
		.select('id, date, merchant, amount, is_private, user_id')
		.eq('family_id', profile.family_id)
		.eq('category_id', categoryId)
		.eq('type', 'out')
		.gte('date', monthStart)
		.lte('date', monthEnd)
		.order('date', { ascending: false });

	const sanitized = (transactions ?? []).map((tx) => ({
		id: tx.id,
		date: tx.date,
		merchant: tx.is_private && tx.user_id !== user.id ? null : tx.merchant,
		amount: Number(tx.amount),
		is_private: tx.is_private
	}));

	return json({ transactions: sanitized });
};
```
